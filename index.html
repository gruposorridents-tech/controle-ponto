// Importando o Firebase v10 Modular direto do CDN import { initializeApp } from "https://www.gstatic.com/firebasejs/10.9.0/firebase-app.js"; import { getFirestore, doc, setDoc, getDoc, collection, getDocs } from "https://www.gstatic.com/firebasejs/10.9.0/firebase-firestore.js"; // SUAS CREDENCIAIS DO FIREBASE (Pegue no Console do Firebase) const firebaseConfig = { apiKey: "SUA_API_KEY", authDomain: "SEU_PROJETO.firebaseapp.com", projectId: "SEU_PROJETO", storageBucket: "SEU_PROJETO.appspot.com", messagingSenderId: "123456789", appId: "SEU_APP_ID" }; // Inicializar Firebase const app = initializeApp(firebaseConfig); const db = getFirestore(app); // Referências da Tela const selectEmpresa = document.getElementById('empresa'); const selectMes = document.getElementById('mes'); const inputAno = document.getElementById('ano'); const inputFuncionario = document.getElementById('nome_funcionario'); const datalistFuncionarios = document.getElementById('lista_funcionarios'); const corpoTabela = document.getElementById('corpo_tabela'); const btnCarregar = document.getElementById('btn_carregar'); const btnSalvar = document.getElementById('btn_salvar'); // Função para formatar o nome em ID (ex: "Danilo Nascimento" -> "danilo_nascimento") const gerarId = (nome) => nome.trim().toLowerCase().replace(/\s+/g, '_'); // ----------------------------------------------------- // FUNÇÃO 1: Buscar Lista de Funcionários da Empresa // ----------------------------------------------------- async function carregarListaFuncionarios() { const empresa = selectEmpresa.value; const funcRef = collection(db, "empresas", empresa, "funcionarios"); datalistFuncionarios.innerHTML = ''; // Limpa lista atual try { const snapshot = await getDocs(funcRef); snapshot.forEach(doc => { const option = document.createElement('option'); option.value = doc.data().nome; // Mostra o nome real datalistFuncionarios.appendChild(option); }); } catch (e) { console.error("Erro ao carregar funcionários:", e); } } // Quando mudar a empresa, recarrega a lista de funcionários selectEmpresa.addEventListener('change', carregarListaFuncionarios); // Carrega na primeira vez que a tela abre carregarListaFuncionarios(); // ----------------------------------------------------- // FUNÇÃO 2: Lógica de Cálculo de Horas (O Motor) // ----------------------------------------------------- function tempoParaMinutos(horaString) { if (!horaString) return 0; const [h, m] = horaString.split(':').map(Number); return (h * 60) + m; } function minutosParaTempo(minutos) { if (minutos <= 0) return "00:00"; const h = Math.floor(minutos / 60); const m = minutos % 60; return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`; } function calcularHoraLinha(event) { // Pega a linha da tabela onde a digitação ocorreu const tr = event.target.closest('tr'); const e1 = tr.querySelector('.e1').value; const s1 = tr.querySelector('.s1').value; const e2 = tr.querySelector('.e2').value; const s2 = tr.querySelector('.s2').value; let totalMinutos = 0; // Cálculo do Primeiro Turno if (e1 && s1) totalMinutos += (tempoParaMinutos(s1) - tempoParaMinutos(e1)); // Cálculo do Segundo Turno if (e2 && s2) totalMinutos += (tempoParaMinutos(s2) - tempoParaMinutos(e2)); // Preenche o total daquele dia tr.querySelector('.total-horas').innerText = minutosParaTempo(totalMinutos); } // ----------------------------------------------------- // FUNÇÃO 3: Gerar Tabela do Mês e Ler do Firebase // ----------------------------------------------------- btnCarregar.addEventListener('click', async () => { const nomeFunc = inputFuncionario.value; if (!nomeFunc) return alert("Digite ou selecione um funcionário!"); const funcId = gerarId(nomeFunc); const empresa = selectEmpresa.value; const ano = inputAno.value; const mes = selectMes.value; // 3.1: Criar Funcionário caso não exista (Conforme seu fluxograma) const docFuncRef = doc(db, "empresas", empresa, "funcionarios", funcId); await setDoc(docFuncRef, { nome: nomeFunc }, { merge: true }); // 3.2: Limpar a tabela e descobrir quantos dias tem o mês corpoTabela.innerHTML = ''; const diasNoMes = new Date(ano, parseInt(mes), 0).getDate(); // 3.3: Buscar dados já salvos (Histórico do Firebase) const chavePonto = `${empresa}_${funcId}_${ano}_${mes}`; const docPontoRef = doc(db, "pontos", chavePonto); const docSnap = await getDoc(docPontoRef); const dadosSalvos = docSnap.exists() ? docSnap.data().dias : {}; // 3.4: Montar Tabela na Tela for (let i = 1; i <= diasNoMes; i++) { const diaDado = dadosSalvos[i] || { e1: '', s1: '', e2: '', s2: '', total: '00:00' }; const tr = document.createElement('tr'); tr.dataset.dia = i; // Guarda o dia na linha para facilitar o salvamento tr.innerHTML = ` ${i} ${diaDado.total} `; // Adiciona evento para recalcular automaticamente ao alterar qualquer input const inputs = tr.querySelectorAll('input'); inputs.forEach(input => input.addEventListener('change', calcularHoraLinha)); corpoTabela.appendChild(tr); } // Mostra o botão de salvar btnSalvar.style.display = 'block'; }); // ----------------------------------------------------- // FUNÇÃO 4: Salvar Tudo no Firebase // ----------------------------------------------------- btnSalvar.addEventListener('click', async () => { const nomeFunc = inputFuncionario.value; const funcId = gerarId(nomeFunc); const empresa = selectEmpresa.value; const ano = inputAno.value; const mes = selectMes.value; // Objeto que vai guardar o ponto de todos os dias const dadosDias = {}; // Lê todas as linhas da tabela da tela const linhas = corpoTabela.querySelectorAll('tr'); linhas.forEach(tr => { const dia = tr.dataset.dia; dadosDias[dia] = { e1: tr.querySelector('.e1').value, s1: tr.querySelector('.s1').value, e2: tr.querySelector('.e2').value, s2: tr.querySelector('.s2').value, total: tr.querySelector('.total-horas').innerText }; }); // Constrói a chave composta const chavePonto = `${empresa}_${funcId}_${ano}_${mes}`; const docPontoRef = doc(db, "pontos", chavePonto); try { // Envia para a nuvem await setDoc(docPontoRef, { empresa, funcionario: funcId, ano, mes, dias: dadosDias }); alert("Ponto salvo com sucesso na nuvem!"); } catch (e) { console.error("Erro ao salvar:", e); alert("Erro ao salvar o ponto."); } });
